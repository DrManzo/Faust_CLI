```python
import json
from dataclasses import dataclass

@dataclass
class ConfigSettings:
    """Data class to hold configuration settings."""
    debug_mode: bool
    max_connections: int
    api_key: str

def load_config(file_path: str) -> ConfigSettings:
    """
    Load and return configuration settings from a JSON file.

    Args:
        file_path (str): The path to the JSON config file.

    Returns:
        ConfigSettings: An instance of ConfigSettings with the loaded configuration.
    """
    with open(file_path, 'r') as f:
        data = json.load(f)
    
    return ConfigSettings(
        debug_mode=data['debug_mode'],
        max_connections=data['max_connections'],
        api_key=data['api_key']
    )
```
